report AccountBalances "Account Balances"
  nest accounts
  by dimension
  date date
  entity accounts_account_balances
  role view_reports
  currency KES
  cache 60
  maxrange 1200

  group account_id
  rows "Account"
  labels accounts_list.record_id > account_name
  sort balance desc

  money_in = sum accounts_money_in.amount "Money In" money
  money_out = sum accounts_money_out.amount "Money Out" money
  transfers_in = sum accounts_transfers.amount where direction=IN "Transfers In" money
  transfers_out = sum accounts_transfers.amount where direction=OUT "Transfers Out" money
  balance = money_in - money_out + transfers_in - transfers_out "Net Movement" money

  headline {label}: {money_in} received, {money_out} spent, {balance} net movement.
  remark balance < 0 | bad | Negative movement | {label} has outgoing movement exceeding incoming movement by {abs(balance)}.
  remark true | neutral | Account movement | {label}: {money_in} received, {money_out} spent, {transfers_in} transferred in and {transfers_out} transferred out.

report AccountMoneyIn "Account Money In"
  nest accounts
  by month|day|year
  date date
  entity accounts_money_in_report
  role view_reports
  currency KES
  cache 60
  maxrange 1200

  filter account_id = account_id

  money_in = sum accounts_money_in.amount "Money In" money
  transactions = count accounts_money_in "Transactions"
  avg_in = money_in / transactions "Average Transaction" money
  chg = change(money_in) "Money In vs Previous"

  headline {label}: {money_in} received across {transactions} transactions[, {updown} {abs(chg)}% on {prev_label}].
  remark transactions == 0 | neutral | No money received | No incoming account transactions were recorded in {label}.
  remark true | good | Money received | {label}: {money_in} received across {transactions} transactions.

  nest accounts
  by month|day|year
  date date
  entity accounts_transfers_report
  role view_reports
  currency KES
  cache 60
  maxrange 1200

  filter account_id = account_id

  transfers_in = sum accounts_transfers.amount where direction=IN "Transfers In" money
  transfers_out = sum accounts_transfers.amount where direction=OUT "Transfers Out" money
  net_transfers = transfers_in - transfers_out "Net Transfers" money

  remark net_transfers < 0 | watch | Net outflow | {label}: accounts transferred out {abs(net_transfers)} more than they received.
  remark true | neutral | Account transfers | {label}: {transfers_in} transferred in and {transfers_out} transferred out.

report AccountForecast "Account Transaction Forecast"
  nest accounts
  by month|year
  date date
  entity accounts_transaction_forecast
  role view_reports
  currency KES
  cache 60
  maxrange 1200

  filter account_id = account_id

  money_in = sum accounts_money_in.amount "Money In" money
  money_out = sum accounts_money_out.amount "Money Out" money
  transfers_in = sum accounts_transfers.amount where direction=IN "Transfers In" money
  transfers_out = sum accounts_transfers.amount where direction=OUT "Transfers Out" money

  net_movement = money_in - money_out + transfers_in - transfers_out "Net Movement" money

  forecast money_in next 12
  forecast money_out next 12
  forecast transfers_in next 12
  forecast transfers_out next 12

  headline {label}: {money_in} received, {money_out} spent, {net_movement} net movement.

  remark money_in == 0 | watch | No money received | No money in was recorded for {label}.
  remark money_out > money_in | watch | Spending exceeds income | Money out exceeded money in for {label}.
  remark true | neutral | Account activity | Historical figures are actual transactions. Forecasts estimate future activity from historical trends.
